package main

import (
	"errors"
	"fmt"
	"testing"
)

func TestHolat(t *testing.T) {
	tests := []struct {
		nom  string
		err  error
		want int
	}{
		{
			nom:  "nil xato - 200",
			err:  nil,
			want: 200,
		},
		{
			nom:  "to'g'ridan-to'g'ri ErrTopilmadi - 404",
			err:  ErrTopilmadi,
			want: 404,
		},
		{
			nom:  "bir marta o'ralgan ErrTopilmadi - 404",
			err:  fmt.Errorf("topish 42: %w", ErrTopilmadi),
			want: 404,
		},
		{
			nom:  "ko'p marta o'ralgan ErrTopilmadi - 404",
			err:  fmt.Errorf("handler: %w", fmt.Errorf("topish 42: %w", ErrTopilmadi)),
			want: 404,
		},
		{
			nom:  "errors.Join ichidagi ErrTopilmadi - 404",
			err:  errors.Join(errors.New("qo'shimcha xato"), ErrTopilmadi),
			want: 404,
		},
		{
			nom:  "boshqa oddiy xato - 500",
			err:  errors.New("ma'lumotlar bazasi ishlamayapti"),
			want: 500,
		},
		{
			nom:  "o'ralgan boshqa xato - 500",
			err:  fmt.Errorf("topish 1: %w", errors.New("ulanish uzildi")),
			want: 500,
		},
		{
			nom:  "%w siz o'ralgan ErrTopilmadi (zanjir uzilgan) - 500",
			err:  fmt.Errorf("topish 42: %v", ErrTopilmadi),
			want: 500,
		},
		{
			nom:  "matni bir xil, lekin boshqa xato qiymati - 500",
			err:  errors.New("buyurtma topilmadi"),
			want: 500,
		},
	}

	for _, tt := range tests {
		t.Run(tt.nom, func(t *testing.T) {
			if got := holat(tt.err); got != tt.want {
				t.Errorf("holat(%v) = %d, kutilgan %d", tt.err, got, tt.want)
			}
		})
	}
}

func TestTopishHolat(t *testing.T) {
	tests := []struct {
		nom  string
		id   string
		want int
	}{
		{nom: "mavjud buyurtma - 200", id: "1", want: 200},
		{nom: "mavjud bo'lmagan buyurtma - 404", id: "42", want: 404},
	}

	for _, tt := range tests {
		t.Run(tt.nom, func(t *testing.T) {
			_, err := topish(tt.id)
			if got := holat(err); got != tt.want {
				t.Errorf("holat(topish(%q)) = %d, kutilgan %d", tt.id, got, tt.want)
			}
		})
	}
}
