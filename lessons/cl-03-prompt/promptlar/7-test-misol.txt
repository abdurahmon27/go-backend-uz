package main

import (
	"errors"
	"fmt"
)

var ErrTopilmadi = errors.New("buyurtma topilmadi")

var buyurtmalar = map[string]string{"1": "olma, 2 kg"}

func topish(id string) (string, error) {
	b, ok := buyurtmalar[id]
	if !ok {
		return "", fmt.Errorf("topish %s: %w", id, ErrTopilmadi)
	}
	return b, nil
}

// holat: handler shu kodni HTTP javobga yozadi
func holat(err error) int {
	if err == nil {
		return 200
	}
	if errors.Is(err, ErrTopilmadi) {
		return 404
	}
	return 500
}

func main() {
	for _, id := range []string{"1", "42"} {
		_, err := topish(id)
		fmt.Printf("GET /buyurtma/%s → %d %v\n", id, holat(err), err)
	}
}
</kod>

package kabisa

import "testing"

func TestJadval(t *testing.T) {
	jadval := []struct {
		nom  string
		yil  int
		want bool
	}{
		{"oddiy", 2026, false},
		{"4 ga", 2024, true},
		{"100 ga", 1900, false},
		{"400 ga", 2000, true},
	}
	for _, tc := range jadval {
		t.Run(tc.nom, func(t *testing.T) {
			if got := kabisa(tc.yil); got != tc.want {
				t.Errorf("kabisa(%d) = %v, kutilgan %v", tc.yil, got, tc.want)
			}
		})
	}
}
