package main

import (
	"errors"
	"fmt"
	"testing"
)

func TestHolat(t *testing.T) {
	boshqa := errors.New("baza ulanmadi")

	jadval := []struct {
		nom  string
		err  error
		want int
	}{
		{"nil xato", nil, 200},
		{"ochiq ErrTopilmadi", ErrTopilmadi, 404},
		{"oʻralgan ErrTopilmadi", fmt.Errorf("topish 42: %w", ErrTopilmadi), 404},
		{"ikki marta oʻralgan", fmt.Errorf("handler: %w", fmt.Errorf("topish 42: %w", ErrTopilmadi)), 404},
		{"boshqa xato", boshqa, 500},
		{"oʻralgan boshqa xato", fmt.Errorf("topish 7: %w", boshqa), 500},
		{"matni bir xil, lekin boshqa xato", errors.New("buyurtma topilmadi"), 500},
		{"%w oʻrniga %v bilan oʻralgan", fmt.Errorf("topish 42: %v", ErrTopilmadi), 500},
	}
	for _, tc := range jadval {
		t.Run(tc.nom, func(t *testing.T) {
			if got := holat(tc.err); got != tc.want {
				t.Errorf("holat(%v) = %d, kutilgan %d", tc.err, got, tc.want)
			}
		})
	}
}
