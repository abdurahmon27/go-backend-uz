package main

import (
	"errors"
	"fmt"
	"testing"
)

func TestHolat(t *testing.T) {
	jadval := []struct {
		nom  string
		xato error
		want int
	}{
		{"xato yoʻq", nil, 200},
		{"ochiq ErrTopilmadi", ErrTopilmadi, 404},
		{"oʻralgan ErrTopilmadi", fmt.Errorf("topish 42: %w", ErrTopilmadi), 404},
		{"ikki marta oʻralgan", fmt.Errorf("handler: %w", fmt.Errorf("topish 42: %w", ErrTopilmadi)), 404},
		{"boshqa xato", errors.New("baza ishlamayapti"), 500},
		{"oʻralgan boshqa xato", fmt.Errorf("topish 1: %w", errors.New("ulanish uzildi")), 500},
		{"matni bir xil, lekin ErrTopilmadi emas", errors.New("buyurtma topilmadi"), 500},
		{"%w oʻrniga %v bilan oʻralgan", fmt.Errorf("topish 42: %v", ErrTopilmadi), 500},
	}
	for _, tc := range jadval {
		t.Run(tc.nom, func(t *testing.T) {
			if got := holat(tc.xato); got != tc.want {
				t.Errorf("holat(%v) = %d, kutilgan %d", tc.xato, got, tc.want)
			}
		})
	}
}
